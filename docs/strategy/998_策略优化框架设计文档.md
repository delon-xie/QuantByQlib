## 4. 优化算法实现

### 4.1 多目标优化函数

```python
# objective_functions.py
class MultiObjectiveOptimizer:
    """多目标优化器"""
    
    def __init__(self, objectives: List[str], weights: Dict = None):
        self.objectives = objectives
        self.weights = weights or {
            'sharpe_ratio': 0.3,
            'max_drawdown': 0.25,
            'total_return': 0.2,
            'win_rate': 0.15,
            'profit_factor': 0.1
        }
    
    def calculate_score(
        self,
        metrics: BacktestMetrics,
        market_regime: str = None
    ) -> float:
        """计算综合得分"""
        
        # 根据市场状态调整权重
        adjusted_weights = self.adjust_weights_for_regime(
            self.weights, market_regime
        )
        
        # 归一化各项指标
        normalized_scores = {
            'sharpe': self.normalize_sharpe(metrics.sharpe_ratio),
            'drawdown': self.normalize_drawdown(metrics.max_drawdown),
            'return': self.normalize_return(metrics.total_return),
            'win_rate': metrics.win_rate,
            'profit_factor': self.normalize_profit_factor(metrics.profit_factor)
        }
        
        # 加权求和
        total_score = sum(
            normalized_scores[obj] * adjusted_weights[obj]
            for obj in self.weights
        )
        
        return total_score
    
    def adjust_weights_for_regime(
        self,
        base_weights: Dict,
        regime: str
    ) -> Dict:
        """根据市场状态调整权重"""
        regime_adjustments = {
            "bull": {'return': 0.3, 'drawdown': 0.2, 'sharpe': 0.25},
            "bear": {'drawdown': 0.4, 'sharpe': 0.3, 'return': 0.1},
            "consolidation": {'win_rate': 0.25, 'sharpe': 0.3, 'drawdown': 0.25}
        }
        
        if regime in regime_adjustments:
            adjustments = regime_adjustments[regime]
            return {**base_weights, **adjustments}
        
        return base_weights
```

### 4.2 贝叶斯优化实现

```python
# bayesian_optimizer.py
class BayesianOptimizer:
    """贝叶斯优化器 - 用于高维参数空间"""
    
    def __init__(
        self,
        strategy_func,
        param_bounds: Dict,
        n_iter: int = 50,
        init_points: int = 5
    ):
        self.strategy_func = strategy_func
        self.param_bounds = param_bounds
        self.n_iter = n_iter
        self.init_points = init_points
        
    def optimize(
        self,
        data: pd.DataFrame,
        metric_func: Callable
    ) -> Dict:
        """运行贝叶斯优化"""
        from skopt import gp_minimize
        from skopt.space import Real, Integer, Categorical
        
        # 定义参数空间
        dimensions = []
        for param_name, bounds in self.param_bounds.items():
            if isinstance(bounds[0], (int, np.integer)):
                dimensions.append(Integer(*bounds, name=param_name))
            elif isinstance(bounds[0], float):
                dimensions.append(Real(*bounds, name=param_name))
            elif isinstance(bounds[0], str):
                dimensions.append(Categorical(bounds, name=param_name))
        
        # 目标函数
        def objective(params):
            param_dict = dict(zip(self.param_bounds.keys(), params))
            strategy = self.strategy_func(**param_dict)
            metrics = self._evaluate_strategy(strategy, data)
            return -metric_func(metrics)  # 负号因为gp_minimize是最小化
        
        # 运行优化
        result = gp_minimize(
            objective,
            dimensions,
            n_calls=self.n_iter,
            n_initial_points=self.init_points,
            random_state=42
        )
        
        best_params = dict(zip(self.param_bounds.keys(), result.x))
        return {
            'best_params': best_params,
            'best_score': -result.fun,  # 转回正值
            'optimization_history': result
        }
```

```python
# robustness_checker.py
class RobustnessChecker:
    """稳健性检验器"""
    
    def __init__(self, n_simulations: int = 1000):
        self.n_simulations = n_simulations
    
    def monte_carlo_simulation(
        self,
        strategy,
        data: pd.DataFrame,
        noise_level: float = 0.01
    ) -> Dict:
        """蒙特卡洛模拟"""
        results = []
        
        for i in range(self.n_simulations):
            # 添加随机噪声
            noisy_data = self.add_random_noise(data, noise_level)
            
            # 运行回测
            metrics = self._run_backtest(strategy, noisy_data)
            results.append(metrics)
        
        # 统计分析
        return self.analyze_simulation_results(results)
    
    def add_random_noise(
        self,
        data: pd.DataFrame,
        noise_level: float
    ) -> pd.DataFrame:
        """添加随机噪声"""
        noisy_data = data.copy()
        
        for col in ['$close', '$high', '$low']:
            if col in noisy_data.columns:
                noise = np.random.normal(0, noise_level, len(noisy_data))
                noisy_data[col] = noisy_data[col] * (1 + noise)
        
        return noisy_data
    
    def parameter_perturbation_test(
        self,
        base_params: Dict,
        strategy_func,
        data: pd.DataFrame,
        perturbation: float = 0.1
    ) -> Dict:
        """参数扰动测试"""
        results = {}
        
        for param_name, base_value in base_params.items():
            if isinstance(base_value, (int, float)):
                # 测试±10%扰动
                perturbations = [
                    base_value * (1 - perturbation),
                    base_value * (1 + perturbation)
                ]
                
                param_results = []
                for perturbed_value in perturbations:
                    params = base_params.copy()
                    params[param_name] = perturbed_value
                    
                    strategy = strategy_func(**params)
                    metrics = self._run_backtest(strategy, data)
                    param_results.append(metrics)
                
                # 计算参数敏感性
                sensitivity = self.calculate_parameter_sensitivity(
                    base_params[param_name], param_results
                )
                results[param_name] = sensitivity
        
        return results
```
