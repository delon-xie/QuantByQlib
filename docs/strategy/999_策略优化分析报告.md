### 4.4 核心逻辑改进建议

#### 4.4.1 高点识别优化

**问题**: 当前 `_find_price_peaks` 方法仅找局部最大值，可能遗漏真正的趋势高点。

**改进方案**:
```python
def _find_price_peaks_enhanced(self, prices: pd.Series, lookback: int = 20, min_distance: int = 3) -> List[int]:
    """改进的高点识别：考虑高点之间的距离和相对高度"""
    peaks = []
    
    for i in range(min_distance, min(lookback, len(prices) - min_distance)):
        idx = len(prices) - i - 1
        
        # 局部最大值
        is_local_max = True
        for j in range(1, min_distance + 1):
            if prices.iloc[idx] <= prices.iloc[idx-j] or prices.iloc[idx] <= prices.iloc[idx+j]:
                is_local_max = False
                break
        
        if is_local_max:
            # 检查是否是重要高点（高于前后一定范围）
            surrounding_max = max(prices.iloc[idx-min_distance:idx+min_distance+1])
            if prices.iloc[idx] >= surrounding_max * 0.95:  # 接近局部最高
                peaks.append(idx)
    
    # 排序并去重（保留最重要的高点）
    peaks.sort(reverse=True)
    return peaks
```

#### 4.4.2 趋势线计算优化

**问题**: 当前仅用最后2个点计算趋势线，稳定性差。

**改进方案**:
```python
def _calculate_trendline_robust(self, trend_points: List[tuple], x_position: int) -> Optional[float]:
    """使用线性回归计算趋势线，提高稳定性"""
    if len(trend_points) < 2:
        return None
    
    x_coords = np.array([p[0] for p in trend_points])
    y_coords = np.array([p[1] for p in trend_points])
    
    # 使用最小二乘法计算趋势线
    if len(trend_points) >= 3:
        # 线性回归
        slope, intercept = np.polyfit(x_coords, y_coords, 1)
        return slope * x_position + intercept
    else:
        # 两点连线（回退方案）
        p1, p2 = trend_points[0], trend_points[-1]
        if p2[0] != p1[0]:
            k = (p2[1] - p1[1]) / (p2[0] - p1[0])
            b = p1[1] - k * p1[0]
            return k * x_position + b
    
    return None
```

#### 4.4.3 突破质量评估增强

```python
def _calculate_breakout_score_enhanced(self, break_ratio: float, trend_slope: float,
                                      close_prices: pd.Series, volume: pd.Series,
                                      trendline_value: float) -> float:
    """增强版突破质量评分"""
    score = 0
    
    # 1. 突破幅度（权重35%）
    score += min(break_ratio * 100, 35)
    
    # 2. 原趋势强度（权重25%）
    trend_strength = abs(trend_slope) * 10000
    score += min(trend_strength, 25)
    
    # 3. 突破前蓄势（波动率收缩）（权重10%）
    if len(close_prices) >= 20:
        recent_volatility = close_prices.pct_change().std() * 100
        if recent_volatility < 2:
            score += 10
    
    # 4. 突破K线形态（权重10%）
    if len(close_prices) >= 3:
        today_change = (close_prices.iloc[-1] - close_prices.iloc[-2]) / close_prices.iloc[-2]
        if today_change > 0.02:
            score += 10
    
    # 5. 成交量配合（权重10%）
    if volume is not None and len(volume) >= 5:
        vol_ratio = volume.iloc[-1] / volume.iloc[-5:].mean()
        if vol_ratio > 1.5:
            score += 10
    
    # 6. 突破后回踩验证（新增，权重10%）
    if self.retest_confirmation:
        if self._check_retest_enhanced(close_prices, trendline_value):
            score += 10
    
    return max(score, 0)
```

### 6.2 优化目标函数

```python
from dataclasses import dataclass

@dataclass
class BacktestMetrics:
    """回测指标"""
    total_return: float           # 总收益率
    sharpe_ratio: float           # 夏普比率
    max_drawdown: float           # 最大回撤
    win_rate: float              # 胜率
    profit_loss_ratio: float     # 盈亏比
    calmar_ratio: float          # 卡玛比率（年化收益/最大回撤）
    sortino_ratio: float         # 索提诺比率
    trade_count: int             # 交易次数
    avg_holding_days: float      # 平均持仓天数

def objective_function(metrics: BacktestMetrics) -> float:
    """
    多目标优化函数
    
    权重可根据投资者偏好调整：
    - 保守型：更看重回撤控制（max_drawdown权重高）
    - 激进型：更看重收益（total_return权重高）
    - 平衡型：综合考虑各项指标
    """
    weights = {
        'total_return': 0.25,     # 收益权重
        'sharpe_ratio': 0.25,    # 风险调整收益
        'max_drawdown': 0.25,    # 回撤控制
        'calmar_ratio': 0.15,    # 收益回撤比
        'win_rate': 0.10,        # 胜率
    }
    
    # 归一化处理
    score = (
        weights['total_return'] * normalize_return(metrics.total_return) +
        weights['sharpe_ratio'] * normalize_sharpe(metrics.sharpe_ratio) +
        weights['max_drawdown'] * normalize_drawdown(metrics.max_drawdown) +
        weights['calmar_ratio'] * normalize_calmar(metrics.calmar_ratio) +
        weights['win_rate'] * normalize_winrate(metrics.win_rate)
    )
    
    return score

def normalize_return(ret: float) -> float:
    """收益率归一化：0%~100% -> 0~1"""
    return min(max(ret / 0.5, 0), 1)  # 50%收益映射为1

def normalize_sharpe(sharpe: float) -> float:
    """夏普比率归一化"""
    return min(max(sharpe / 2.0, 0), 1)  # 2.0以上为满分

def normalize_drawdown(dd: float) -> float:
    """回撤归一化：越小越好"""
    return 1 - min(abs(dd) / 0.3, 1)  # 30%回撤映射为0

def normalize_calmar(calmar: float) -> float:
    """卡玛比率归一化"""
    return min(max(calmar / 1.5, 0), 1)

def normalize_winrate(wr: float) -> float:
    """胜率归一化"""
    return wr  # 胜率本身就是0~1
```

```python
# parameter_optimizer.py
from itertools import product
from tqdm import tqdm

class ParameterOptimizer:
    """参数优化器"""
    
    def __init__(self, strategy, param_grid: Dict, metric_func):
        self.strategy = strategy
        self.param_grid = param_grid
        self.metric_func = metric_func
    
    def grid_search(
        self,
        train_data,
        n_jobs: int = -1,
    ) -> Dict:
        """
        网格搜索
        
        Args:
            train_data: 训练数据
            n_jobs: 并行任务数
        
        Returns:
            最佳参数组合和对应的得分
        """
        # 生成参数组合
        param_names = list(self.param_grid.keys())
        param_values = list(self.param_grid.values())
        param_combinations = list(product(*param_values))
        
        results = []
        
        for params in tqdm(param_combinations):
            param_dict = dict(zip(param_names, params))
            
            # 创建策略实例
            strategy = self.strategy(**param_dict)
            
            # 运行回测
            signals = strategy.generate_signals(train_data)
            metrics = BacktestEngine().run(signals, train_data)
            
            # 计算目标函数
            score = self.metric_func(metrics)
            
            results.append({
                'params': param_dict,
                'metrics': metrics,
                'score': score,
            })
        
        # 返回最佳结果
        best = max(results, key=lambda x: x['score'])
        return best
```

### 7.3 第三阶段：Walk-Forward 分析

```python
# walk_forward.py
class WalkForwardOptimizer:
    """Walk-Forward 分析"""
    
    def __init__(
        self,
        strategy,
        param_grid: Dict,
        train_window: int = 252,   # 训练窗口（交易日）
        test_window: int = 63,      # 测试窗口（季度）
        step: int = 21,             # 滚动步长（月度）
    ):
        self.strategy = strategy
        self.param_grid = param_grid
        self.train_window = train_window
        self.test_window = test_window
        self.step = step
    
    def run(self, prices: pd.DataFrame) -> Dict:
        """运行Walk-Forward分析"""
        
        results = []
        dates = prices.index
        
        i = 0
        while i + self.train_window + self.test_window <= len(dates):
            # 划分训练集和测试集
            train_start = dates[i]
            train_end = dates[i + self.train_window]
            test_start = dates[i + self.train_window]
            test_end = dates[i + self.train_window + self.test_window]
            
            train_data = prices[train_start:train_end]
            test_data = prices[test_start:test_end]
            
            # 训练阶段：网格搜索找最佳参数
            optimizer = ParameterOptimizer(
                self.strategy, 
                self.param_grid,
                self.metric_func
            )
            best_params = optimizer.grid_search(train_data)
            
            # 测试阶段：用最佳参数运行回测
            strategy = self.strategy(**best_params['params'])
            test_metrics = BacktestEngine().run(
                strategy.generate_signals(test_data),
                test_data
            )
            
            results.append({
                'train_period': (str(train_start), str(train_end)),
                'test_period': (str(test_start), str(test_end)),
                'best_params': best_params['params'],
                'train_score': best_params['score'],
                'test_metrics': test_metrics,
            })
            
            i += self.step
        
        return self._analyze_results(results)
    
    def _analyze_results(self, results: List) -> Dict:
        """分析Walk-Forward结果"""
        
        train_scores = [r['train_score'] for r in results]
        test_returns = [r['test_metrics'].total_return for r in results]
        test_sharpes = [r['test_metrics'].sharpe_ratio for r in results]
        
        return {
            'avg_train_score': np.mean(train_scores),
            'avg_test_return': np.mean(test_returns),
            'avg_test_sharpe': np.mean(test_sharpes),
            'test_return_std': np.std(test_returns),
            'sharpe_std': np.std(test_sharpes),
            'param_stability': self._check_param_stability(results),
        }
```

```python
# sensitivity_analysis.py
class SensitivityAnalyzer:
    """敏感性分析"""
    
    def __init__(self, strategy, base_params: Dict, metric_func):
        self.strategy = strategy
        self.base_params = base_params
        self.metric_func = metric_func
    
    def analyze(
        self,
        param_name: str,
        data,
        range_factor: float = 0.5,  # ±50%范围
        steps: int = 20,
    ) -> pd.DataFrame:
        """
        分析单个参数的敏感性
        
        Args:
            param_name: 参数名
            data: 历史数据
            range_factor: 相对于基准值的变化范围
            steps: 扫描点数
        
        Returns:
            敏感性分析结果
        """
        base_value = self.base_params[param_name]
        
        # 生成参数范围
        lower = base_value * (1 - range_factor)
        upper = base_value * (1 + range_factor)
        param_range = np.linspace(lower, upper, steps)
        
        results = []
        
        for value in param_range:
            params = self.base_params.copy()
            params[param_name] = value
            
            strategy = self.strategy(**params)
            metrics = BacktestEngine().run(
                strategy.generate_signals(data),
                data
            )
            
            results.append({
                'param_value': value,
                'score': self.metric_func(metrics),
                'return': metrics.total_return,
                'sharpe': metrics.sharpe_ratio,
                'drawdown': metrics.max_drawdown,
            })
        
        return pd.DataFrame(results)
```
