### 阶段4：参数优化核心

**文件**: `optimization/optimizer.py`

#### 4.1 网格搜索实现

```python
class GridSearchOptimizer(OptimizerBase):
    def optimize(self, executor: StrategyExecutor) -> dict:
        """遍历所有参数组合"""
        all_params = self._generate_grid()  # 生成500+组参数
        best_score = -float('inf')
        best_params = None
        
        for i, params in enumerate(all_params):
            # 进度更新
            self._callback(i / len(all_params))
            
            # 执行策略并评分
            score = executor.execute(params)
            
            # 更新最优解
            if score > best_score:
                best_score = score
                best_params = params
        
        return {
            "best_params": best_params,
            "best_score": best_score,
            "total_iterations": len(all_params)
        }
```

#### 4.2 策略执行器

**文件**: `optimization/strategy_executor.py`

```python
def execute(self, params: dict) -> float:
    """执行策略并返回评分"""
    # 1. 加载策略类
    strategy_class = self._get_strategy_class()  # EarlyTrendStrategy
    
    # 2. 初始化策略实例
    strategy = strategy_class(**params)
    
    # 3. 运行策略（回测）
    result = strategy.run(
        universe=self.universe,
        prices=self.prices,
        start_date=self.start_date,
        end_date=self.end_date
    )
    
    # 4. 多目标评分
    score = self._calculate_score(result)
    return score
```

#### 4.3 多目标评分函数

```python
def _calculate_score(self, result) -> float:
    """综合评分：收益率 + 夏普比率 + 最大回撤"""
    weights = {
        "return": 0.4,      # 收益率权重
        "sharpe": 0.3,      # 夏普比率权重
        "drawdown": 0.3     # 最大回撤权重
    }
    
    return (
        weights["return"] * result.annual_return +
        weights["sharpe"] * result.sharpe_ratio +
        weights["drawdown"] * (1 + result.max_drawdown)  # 转换为正值
    )
```

### 阶段5：Walk-Forward 验证

**核心思想**：将历史数据划分为多个滚动窗口，验证参数稳定性

```python
def _walk_forward_validation(self, best_params: dict) -> dict:
    """Walk-Forward交叉验证"""
    windows = self._split_into_windows(n_windows=5)  # 5个时间窗口
    
    train_scores = []
    test_scores = []
    
    for train_period, test_period in windows:
        # 在训练期重新优化
        train_score = self._optimize_on_period(best_params, train_period)
        train_scores.append(train_score)
        
        # 在测试期验证
        test_score = self._evaluate_on_period(best_params, test_period)
        test_scores.append(test_score)
    
    return {
        "n_windows": 5,
        "avg_train_score": np.mean(train_scores),
        "avg_test_score": np.mean(test_scores),
        "test_score_std": np.std(test_scores),
        "param_stability": self._calculate_stability()
    }
```

### 阶段6：稳健性检验

#### 6.1 参数敏感性分析

```python
def _parameter_sensitivity(self, best_params: dict) -> dict:
    """分析每个参数对策略表现的影响"""
    results = {}
    
    for param_name, value in best_params.items():
        # 对每个参数进行扰动（±10%）
        original_score = self._evaluate(best_params)
        
        perturbed_params = best_params.copy()
        perturbed_params[param_name] = value * 1.1
        score_up = self._evaluate(perturbed_params)
        
        perturbed_params[param_name] = value * 0.9
        score_down = self._evaluate(perturbed_params)
        
        # 计算敏感度
        sensitivity = abs(score_up - score_down) / (0.2 * original_score)
        results[param_name] = {
            "base_value": value,
            "sensitivity": sensitivity
        }
    
    return results
```

#### 6.2 蒙特卡洛模拟

```python
def _monte_carlo_simulation(self, best_params: dict) -> dict:
    """随机扰动股票池，检验策略稳健性"""
    n_simulations = 1000
    scores = []
    
    for _ in range(n_simulations):
        # 随机抽样股票池（80%）
        sample_universe = np.random.choice(
            self.universe, 
            size=int(len(self.universe) * 0.8),
            replace=False
        )
        
        # 评估策略
        score = self._evaluate(best_params, universe=sample_universe)
        scores.append(score)
    
    return {
        "n_simulations": n_simulations,
        "score_mean": np.mean(scores),
        "score_std": np.std(scores),
        "score_p5": np.percentile(scores, 5),
        "score_p95": np.percentile(scores, 95),
        "pass_threshold": np.mean(scores) > 0.3
    }
```

## 五、当前问题分析

**段错误位置**：网格搜索第1/500次迭代

**可能原因**：
1. **策略执行器内存问题** — `StrategyExecutor.execute()` 内部访问非法内存
2. **数据异常** — 某支股票数据包含NaN或极端值
3. **多进程资源泄漏** — joblib警告提示

**解决方案**：
```python
# 临时修复：添加异常捕获
def execute(self, params: dict) -> float:
    try:
        # 原有逻辑
        strategy = strategy_class(**params)
        result = strategy.run(...)
        return self._calculate_score(result)
    except Exception as e:
        logger.warning(f"策略执行失败: {e}")
        return -float('inf')  # 返回最低分
```
